-- check-qa.sh
#!/usr/bin/env bash
echo "=========================================================="
echo "          ESTADO DE SALUD DEL AMBIENTE QA (10.43.100.141)"
echo "=========================================================="

echo -e "\n1. Contenedores en ejecucion:"
docker compose -f /opt/mani/docker-compose.qa.yml ps

echo -e "\n2. Verificacion de Endpoints:"
endpoints=(
    "Gateway Health|http://localhost/health"
    "Core Service Health|http://localhost/api/v1/core/health"
    "Rules Service Health|http://localhost/api/v1/rules/health"
    "Dispatch Service Health|http://localhost/api/v1/dispatch/health"
    "Flutter Web|http://localhost:8080/"
    "Landing Web|http://localhost:3001/"
)

for item in "${endpoints[@]}"; do
    IFS="|" read -r name url <<< "$item"
    code=$(curl -s -o /dev/null -w "%{http_code}" -m 5 "$url")
    body=$(curl -s -m 5 "$url" | head -c 90 | tr '\n' ' ')
    if [ "$code" == "200" ]; then
        echo -e "  [\e[32m$code OK\e[0m] $name -> $body"
    else
        echo -e "  [\e[31m$code FAIL\e[0m] $name -> $body"
    fi
done
echo "=========================================================="
